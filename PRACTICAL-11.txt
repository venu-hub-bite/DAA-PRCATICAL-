
#include <iostream>
using namespace std;

int main() {
    int n, d[10][10];
    cin >> n;

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> d[i][j];

    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i][k] != 999 && d[k][j] != 999)
                    d[i][j] = min(d[i][j],
                                  d[i][k] + d[k][j]);

    cout << "Shortest distance matrix:\n";
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++)
            cout << d[i][j] << " ";
        cout << endl;
    }
    return 0;
}
