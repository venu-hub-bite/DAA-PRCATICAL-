
#include <iostream>
using namespace std;

int main() {
    int a[100], n, key, choice;
    cout << "Enter number of elements: ";
    cin >> n;

    cout << "Enter elements: ";
    for (int i = 0; i < n; i++)
        cin >> a[i];

    cout << "\n1. Linear Search";
    cout << "\n2. Binary Search";
    cout << "\nEnter your choice: ";
    cin >> choice;

    cout << "Enter element to search: ";
    cin >> key;

    int pos = -1;

    if (choice == 1) {
        // Linear Search
        for (int i = 0; i < n; i++) {
            if (a[i] == key) {
                pos = i;
                break;
            }
        }
    }
    else if (choice == 2) {
        // Binary Search (array must be sorted)
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - i - 1; j++) {
                if (a[j] > a[j + 1]) {
                    int temp = a[j];
                    a[j] = a[j + 1];
                    a[j + 1] = temp;
                }
            }
        }

        int low = 0, high = n - 1;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            if (a[mid] == key) {
                pos = mid;
                break;
            }
            else if (key < a[mid])
                high = mid - 1;
            else
                low = mid + 1;
        }
    }
    else {
        cout << "Invalid choice!";
        return 0;
    }

    if (pos != -1)
        cout << "Element found at position " << pos + 1;
    else
        cout << "Element not found";

    return 0;
}
