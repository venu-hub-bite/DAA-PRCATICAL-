#include <iostream>
#include <vector>
#include <queue>
using namespace std;

vector<int> adj[100];
bool visited[100];

// Depth First Search
void DFS(int v) {
    visited[v] = true;
    cout << v << " ";

    for (int u : adj[v])
        if (!visited[u])
            DFS(u);
}

// Breadth First Search
void BFS(int start, int n) {
    fill(visited, visited + n, false);
    queue<int> q;
    q.push(start);
    visited[start] = true;

    while (!q.empty()) {
        int v = q.front();
        q.pop();
        cout << v << " ";

        for (int u : adj[v]) {
            if (!visited[u]) {
                visited[u] = true;
                q.push(u);
            }
        }
    }
}

int main() {
    int n, e, u, v;
    cout << "Enter vertices and edges: ";
    cin >> n >> e;

    cout << "Enter edges:\n";
    for (int i = 0; i < e; i++) {
        cin >> u >> v;
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    int start;
    cout << "Enter starting vertex: ";
    cin >> start;

    cout << "DFS: ";
    fill(visited, visited + n, false);
    DFS(start);

    cout << "\nBFS: ";
    BFS(start, n);

    return 0;
}
