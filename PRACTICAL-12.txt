
#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n, a[10][10], p[10], cost, minCost = 999999;

    cin >> n;

    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> a[i][j];

    for (int i = 0; i < n; i++)
        p[i] = i;

    do {
        cost = 0;
        for (int i = 0; i < n - 1; i++)
            cost += a[p[i]][p[i + 1]];

        cost += a[p[n - 1]][p[0]];

        minCost = min(minCost, cost);
    } while (next_permutation(p + 1, p + n));

    cout << "Minimum cost = " << minCost;
    return 0;
}
