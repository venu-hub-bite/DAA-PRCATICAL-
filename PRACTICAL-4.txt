
#include <iostream>
using namespace std;

// Iterative method
long long factorialIterative(int n) {
    long long fact = 1;
    for (int i = 1; i <= n; i++)
        fact *= i;
    return fact;
}

// Recursive method
long long factorialRecursive(int n) {
    if (n <= 1)
        return 1;
    return n * factorialRecursive(n - 1);
}

int main() {
    int n;
    cout << "Enter a number (0 to 20): ";
    cin >> n;

    if (n < 0 || n > 20) {
        cout << "Invalid input!";
        return 0;
    }

    cout << "Factorial using Iteration: "
         << factorialIterative(n) << endl;

    cout << "Factorial using Recursion: "
         << factorialRecursive(n) << endl;

    return 0;
}
